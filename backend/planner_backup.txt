from __future__ import annotations

import json
import uuid

from app.services.agents.base_agent import BaseAgent

from app.core.ai_engine.context import AIContext
from app.core.execution.ai_task import AITask
from app.core.execution.task_priority import TaskPriority
from app.core.execution.task_status import TaskStatus
from app.core.planner.execution_plan import ExecutionPlan

from app.schemas.planner import (
    PlannerResponse,
    PlannedTask,
)

from app.services.llm_service import LLMService


class PlannerAgent(BaseAgent):

    @property
    def name(self) -> str:
        return "PlannerAgent"

    @property
    def capabilities(self) -> list[str]:
        return ["planning"]

    async def execute(
        self,
        context: AIContext,
    ):

        # =====================================================
        # Capability Registry
        # =====================================================

        if context.capability_registry is None:
            raise ValueError(
                "Capability registry is not available."
            )

        capabilities = (
            context.capability_registry
            .list_capabilities()
        )

        # =====================================================
        # Dataset Information
        # =====================================================

        dataset_info = "No dataset uploaded."

        if context.dataset is not None:

            dataset_info = f"""
Dataset columns:

{", ".join(
    str(column)
    for column in context.dataset.columns
)}

Rows: {len(context.dataset)}

Columns: {len(context.dataset.columns)}
"""

        # =====================================================
        # Planner Prompt
        # =====================================================

        prompt = f"""
You are the planning engine of DataNova AI.

Your job is to convert the user's request into
a sequence of executable tasks.

AVAILABLE CAPABILITIES:

{json.dumps(
    capabilities,
    indent=2
)}

DATASET INFORMATION:

{dataset_info}

USER REQUEST:

{context.message}

RULES:

1. Only use capabilities from the available capability list.

2. Never invent a capability.

3. Create only the tasks necessary to satisfy the request.

4. Put tasks in logical execution order.

5. Use HIGH priority for core analysis and machine learning tasks.

6. Use MEDIUM priority for visualization and report tasks.

7. Use LOW priority for optional tasks.

8. If a dataset is required but unavailable,
   still create the appropriate task.

9. If the user asks to explain, describe, summarize,
   analyze, explore, or give an overview of an uploaded
   dataset, use analyze_dataset.

10. If train_model is selected, evaluate_model MUST
    be executed after training.

11. Do NOT add analyze_dataset before train_model unless
    the user explicitly asks for dataset analysis.

12. Do NOT add analyze_dataset before visualization tasks
    unless the user explicitly asks for dataset analysis.

13. Do NOT add generate_report unless the user explicitly
    asks for a report.

14. For histogram and boxplot, identify the requested column.

15. For scatter, identify x and y columns when possible.

16. For bar_chart, identify x and y columns when possible.

17. For line_chart, identify x and y columns when possible.

18. For pie_chart, identify the requested categorical column.

19. For correlation and heatmap, use numerical columns when
    appropriate.

20. For percentage, proportion, or "share" requests involving
    one categorical column, use pie_chart and use that column
    as the pie chart column. Do NOT invent a numerical y-axis.

21. For requests such as "distribution of passengers by Sex",
    use bar_chart with Sex as the categorical column unless
    the user explicitly asks for percentages, proportions,
    or a pie chart.

22. For requests such as "relationship between Age and Fare",
    use scatter with Age as x and Fare as y. Do NOT add a
    correlation task unless the user explicitly asks for
    correlation or a correlation matrix.

23. For generate_python, input_data MUST contain:

{{
    "prompt": "complete description of the Python code to generate"
}}

24. For generate_sql, input_data MUST contain:

{{
    "prompt": "complete description of the SQL to generate"
}}

25. Every task MUST contain an action.

26. Return ONLY valid JSON.

The response MUST have this structure:

{{
    "objective": "string",
    "tasks": [
        {{
            "capability": "string",
            "action": "string",
            "priority": "HIGH",
            "input_data": {{}}
        }}
    ]
}}

Return ONLY JSON.
"""

        # =====================================================
        # Call LLM
        # =====================================================

        response = LLMService.generate(
            prompt
        )

        if not response.success:
            raise RuntimeError(
                response.message
            )

        # =====================================================
        # Parse JSON
        # =====================================================

        raw = response.data.strip()

        if raw.startswith("```"):

            raw = (
                raw
                .replace("```json", "")
                .replace("```", "")
                .strip()
            )

        try:

            planner_data = json.loads(
                raw
            )

        except json.JSONDecodeError as e:

            raise ValueError(
                f"Planner returned invalid JSON: {e}\n"
                f"Response: {raw}"
            )

        # =====================================================
        # Validate Planner Response
        # =====================================================

        planner_response = (
            PlannerResponse.model_validate(
                planner_data
            )
        )

        # =====================================================
        # USER MESSAGE
        # =====================================================

        user_message = (
            context.message
            .lower()
            .strip()
        )

        # =====================================================
        # MACHINE LEARNING INTENT
        # =====================================================

        ml_keywords = [

            "train a model",
            "train model",
            "train machine learning",
            "machine learning model",

            "build a model",
            "build model",

            "fit a model",
            "fit model",

            "machine learning",

            "classification",
            "classifier",

            "regression",
            "regressor",

            "predict",
            "prediction",
            "predictions",

            "forecast",
            "forecasting",

        ]

        ml_requested = any(
            keyword in user_message
            for keyword in ml_keywords
        )

        # =====================================================
        # EVALUATION INTENT
        # =====================================================

        evaluation_keywords = [

            "evaluate the model",
            "evaluate model",

            "evaluate the trained model",
            "evaluate trained model",

            "evaluate my model",

            "model evaluation",

            "evaluate performance",
            "evaluate its performance",

            "check model performance",
            "check the model performance",

            "assess model performance",
            "assess the model",

            "model metrics",
            "show model metrics",

        ]

        evaluation_requested = any(
            keyword in user_message
            for keyword in evaluation_keywords
        )

        # =====================================================
        # DATASET QUESTION INTENT
        # =====================================================

        data_question_keywords = [

            # Counts
            "how many",
            "number of",
            "count of",

            # Statistics
            "average",
            "mean",
            "median",
            "minimum",
            "maximum",
            "lowest",
            "highest",
            "smallest",
            "largest",

            # Dataset structure
            "how many rows",
            "how many columns",
            "what columns",
            "which columns",
            "column names",
            "columns in the dataset",
            "rows in the dataset",

            # Missing data
            "missing values",
            "missing data",
            "null values",
            "nulls",
            "empty values",

            # Duplicates
            "duplicate rows",
            "duplicates",

            # Unique values
            "unique values",
            "unique categories",
            "distinct values",
        ]

        data_question_requested = any(
            keyword in user_message
            for keyword in data_question_keywords
        )

        # =====================================================
        # DATASET ANALYSIS INTENT
        # =====================================================

        analysis_keywords = [

            # Direct analysis

            "analyze dataset",
            "analyse dataset",

            "analyze the dataset",
            "analyse the dataset",

            # Profiling

            "profile dataset",
            "profile the dataset",

            # Explanation

            "explain this dataset",
            "explain the dataset",
            "explain my dataset",

            "explain this data",
            "explain the data",

            # Dataset description

            "tell me about this dataset",
            "tell me about the dataset",
            "tell me about my dataset",

            "describe this dataset",
            "describe the dataset",

            # Summary

            "summarize this dataset",
            "summarise this dataset",

            "summarize the dataset",
            "summarise the dataset",

            "summary of the dataset",

            # Overview

            "dataset overview",
            "dataset summary",

            "give me an overview of the dataset",
            "give me an overview of this dataset",

            "give me an overview",

            # Dataset contents

            "what is in this dataset",
            "what's in this dataset",

            "what is in the dataset",
            "what's in the dataset",

            # Exploration

            "dataset analysis",
            "data analysis",

            "explore dataset",
            "explore the dataset",

            "exploratory data analysis",
            "exploratory analysis",

            "eda",

        ]

        analysis_requested = any(
            keyword in user_message
            for keyword in analysis_keywords
        )

        # =====================================================
        # VISUALIZATION INTENT
        # =====================================================

        visualization_keywords = [

            "scatter plot",
            "scatterplot",
            "scatter",

            "bar chart",
            "bar graph",
            "bar",

            "histogram",

            "line chart",
            "line graph",
            "line",

            "pie chart",
            "pie",

            "boxplot",
            "box plot",

            "heatmap",

            "correlation",
            "correlation matrix",   
            "correlation heatmap",


            # =====================================================
            # General visualization words
            # =====================================================

            "chart",
            "charts",

    "plot",
    "plots",

    "graph",
    "graphs",

    "visualize",
    "visualise",

    "visualization",
    "visualisation",

    # =====================================================
    # Natural-language visualization requests
    # =====================================================

    "distribution",
    "distributions",

    "breakdown",

    "percentage",
    "percentages",

    "proportion",
    "proportions",

    "relationship",

    "relationships",

    "trend",
    "trends",

    "compare",
    "comparison",

    "pattern",
    "patterns",

]

        

        visualization_requested = any(
            keyword in user_message
            for keyword in visualization_keywords
        )

        correlation_keywords = [
    "correlation",
    "correlation matrix",
    "correlations",
]

        

        correlation_requested = any(
            keyword in user_message
            for keyword in correlation_keywords
        )

        # =====================================================
        # PERCENTAGE / PROPORTION INTENT
        # =====================================================

        percentage_keywords = [
            "percentage",
            "percentages",
            "proportion",
            "proportions",
            "percentage of",
            "percent of",
        ]

        percentage_requested = any(
            keyword in user_message
            for keyword in percentage_keywords
        )

        # =====================================================
        # REPORT INTENT
        # =====================================================

        report_keywords = [

            "generate a report",
            "generate report",

            "create a report",
            "create report",

            "make a report",
            "make report",

            "final report",

            "summary report",

            "comprehensive report",

            "analysis report",

            "give me a report",

            "provide a report",

            "prepare a report",

        ]

        report_requested = any(
            keyword in user_message
            for keyword in report_keywords
        )

        # =====================================================
        # CLEAN LLM TASKS
        # =====================================================

        cleaned_tasks = []

        visualization_capabilities = {
            "histogram",
            "scatter",
            "bar_chart",
            "line_chart",
            "pie_chart",
            "boxplot",
            "correlation",
            "heatmap",
        }

       

        for task in planner_response.tasks:
           

            capability = (
                task.capability
                .lower()
                .strip()
            )

            

            # -------------------------------------------------
            # Percentage / proportion requests should use
            # a pie chart when a categorical column is involved.
            # -------------------------------------------------

            if (
                percentage_requested
                and capability == "bar_chart"
            ):
                capability = "pie_chart"
                task.capability = "pie_chart"

            # -------------------------------------------------
            # Prevent correlation from being generated when
            # the user only asks for a relationship/scatter plot
            # -------------------------------------------------

            if (
                 capability == "correlation"
                 and not correlation_requested
                ):               
                continue

            # -------------------------------------------------
            # Prevent general from stealing dataset analysis
            # -------------------------------------------------

            if (
                capability == "general"
                and (
                    analysis_requested
                    or data_question_requested
                )
            ):
                continue

            # -------------------------------------------------
            # Remove unnecessary dataset analysis
            # -------------------------------------------------

            if (
                capability == "analyze_dataset"
                and (
                    ml_requested
                    or visualization_requested
                    or data_question_requested
                )
                and not analysis_requested
            ):
                continue

            # -------------------------------------------------
            # Evaluation
            # -------------------------------------------------

            if (
                capability == "evaluate_model"
                and not evaluation_requested
                and not ml_requested
            ):
                continue

            # -------------------------------------------------
            # Report
            # -------------------------------------------------

            if (
                capability == "generate_report"
                and not report_requested
            ):
                continue

            # -------------------------------------------------
            # Visualization
            # -------------------------------------------------

            if (
                capability in visualization_capabilities
                and not visualization_requested
            ):
                continue

            cleaned_tasks.append(task)

        planner_response.tasks = cleaned_tasks

        # =====================================================
        # FORCE DATASET ANALYSIS
        #
        # If the user clearly asks about the dataset,
        # make sure analyze_dataset exists.
        # =====================================================

        has_analysis_task = any(
            task.capability == "analyze_dataset"
            for task in planner_response.tasks
        )

        if (
            analysis_requested
            and not has_analysis_task
        ):

            planner_response.tasks.insert(
                0,
                PlannedTask(
                    capability="analyze_dataset",
                    action=(
                        "Analyze and explain "
                        "the uploaded dataset"
                    ),
                    priority="HIGH",
                    input_data={},
                )
            )

        # =========================================================
# FORCE DATASET QUESTION
#
# Any question classified as data_question MUST be handled
# by DataQueryAgent.
#
# This is especially important for conversational follow-ups
# such as:
#
#   "What was the value you just calculated?"
#   "What was the previous result?"
#   "What did you just calculate?"
#
# These questions may not explicitly mention a dataset column,
# but they still refer to the previous dataset calculation.
# =========================================================

has_data_question_task = any(
    task.capability == "data_question"
    for task in planner_response.tasks
)

if (
    data_question_requested
    and not analysis_requested
    and not ml_requested
    and not report_requested
    and not visualization_requested
):

    # -----------------------------------------------------
    # Remove incorrectly generated general tasks.
    #
    # The planner's LLM may classify a contextual follow-up
    # as general_chat. When IntentRouter has already decided
    # this is a data_question, do not allow that decision to
    # be overridden by the planner.
    # -----------------------------------------------------

    planner_response.tasks = [
        task
        for task in planner_response.tasks
        if task.capability != "general"
        and task.capability != "general_chat"
    ]

    # -----------------------------------------------------
    # Re-check after removing general tasks.
    # -----------------------------------------------------

    has_data_question_task = any(
        task.capability == "data_question"
        for task in planner_response.tasks
    )

    # -----------------------------------------------------
    # Guarantee exactly one data_question task.
    # -----------------------------------------------------

    if not has_data_question_task:

        planner_response.tasks.insert(
            0,
            PlannedTask(
                capability="data_question",
                action=(
                    "Answer the user's question using "
                    "the uploaded dataset and relevant "
                    "conversation history."
                ),
                priority="HIGH",
                input_data={
                    "question": context.message,
                },
            ),
        )

        # =====================================================
        # FORCE MACHINE LEARNING EVALUATION
        # =====================================================

        has_train_task = any(
            task.capability == "train_model"
            for task in planner_response.tasks
        )

        has_evaluation_task = any(
            task.capability == "evaluate_model"
            for task in planner_response.tasks
        )

        if (
            has_train_task
            and not has_evaluation_task
        ):

            planner_response.tasks.append(
                PlannedTask(
                    capability="evaluate_model",
                    action=(
                        "Evaluate the trained "
                        "machine learning model"
                    ),
                    priority="HIGH",
                    input_data={},
                )
            )

        # =====================================================
        # ML TASK ORDER
        # =====================================================

        if has_train_task:

            train_tasks = [
                task
                for task in planner_response.tasks
                if task.capability == "train_model"
            ]

            evaluation_tasks = [
                task
                for task in planner_response.tasks
                if task.capability == "evaluate_model"
            ]

            other_tasks = [
                task
                for task in planner_response.tasks
                if task.capability
                not in {
                    "train_model",
                    "evaluate_model",
                }
            ]

            planner_response.tasks = (
                train_tasks
                + evaluation_tasks
                + other_tasks
            )

        
                # =====================================================
        # REMOVE DUPLICATE VISUALIZATION TASKS
        # =====================================================

        unique_tasks = []
        seen_visualizations = set()

        visualization_capabilities = {
            "histogram",
            "scatter",
            "bar_chart",
            "line_chart",
            "pie_chart",
            "boxplot",
            "correlation",
            "heatmap",
        }

        for task in planner_response.tasks:

            capability = (
                task.capability
                .lower()
                .strip()
            )

            # Non-visualization tasks are never deduplicated
            if capability not in visualization_capabilities:

                unique_tasks.append(task)
                continue

            input_data = (
                task.input_data
                or {}
            )

            # -------------------------------------------------
            # Build visualization signature
            # -------------------------------------------------

            if capability in {
                "histogram",
                "boxplot",
                "pie_chart",
            }:

                signature = (
                    capability,
                    str(
                        input_data.get(
                            "column",
                            ""
                        )
                    ),
                )

            elif capability in {
                "scatter",
                "line_chart",
            }:

                x = str(
                    input_data.get(
                        "x",
                        ""
                    )
                )

                y = str(
                    input_data.get(
                        "y",
                        ""
                    )
                )

                # Treat X/Y reversal as the same chart
                pair = tuple(
                    sorted(
                        [
                            x,
                            y,
                        ]
                    )
                )

                signature = (
                    capability,
                    pair,
                )

            elif capability == "bar_chart":

                column = str(
                    input_data.get(
                        "column",
                        ""
                    )
                )

                x = str(
                    input_data.get(
                        "x",
                        ""
                    )
                )

                y = str(
                    input_data.get(
                        "y",
                        ""
                    )
                )

                # Frequency bar chart
                if column:

                    signature = (
                        capability,
                        "column",
                        column,
                    )

                # X/Y bar chart
                else:

                    signature = (
                        capability,
                        "xy",
                        x,
                        y,
                    )

            elif capability in {
                "heatmap",
                "correlation",
            }:

                # Only one of each is useful
                signature = (
                    capability,
                )

            else:

                signature = (
                    capability,
                    str(input_data),
                )

            # -------------------------------------------------
            # Skip duplicate
            # -------------------------------------------------

            if signature in seen_visualizations:

                continue

            seen_visualizations.add(
                signature
            )

            unique_tasks.append(task)

        planner_response.tasks = (
            unique_tasks
        )
        # =====================================================
        # CREATE EXECUTION PLAN
        # =====================================================

        plan = ExecutionPlan(
            objective=planner_response.objective,
            tasks=[]
        )
        

        # =====================================================
        # CONVERT PLANNED TASKS → AI TASKS
        # =====================================================

        for planned_task in planner_response.tasks:

            capability = (
                planned_task.capability
                .lower()
                .strip()
            )

            # -------------------------------------------------
            # Validate capability
            # -------------------------------------------------

            if not context.capability_registry.has_capability(
                capability
            ):

                raise ValueError(
                    "Planner selected unknown capability: "
                    f"{capability}"
                )

            # -------------------------------------------------
            # Priority
            # -------------------------------------------------

            priority = self.get_priority(
                planned_task.priority
            )

            # -------------------------------------------------
            # Action
            # -------------------------------------------------

            action = (
                planned_task.action.strip()
                if planned_task.action
                else ""
            )

            if not action:

                action = self.default_action(
                    capability
                )

            # -------------------------------------------------
            # Input data
            # -------------------------------------------------

            input_data = (
                planned_task.input_data
                or {}
            )

                        # =================================================
            # VISUALIZATION INPUT RESOLUTION
            # =================================================

            if context.dataset is not None:

                df = context.dataset

                dataset_columns = [
                    str(column)
                    for column in df.columns
                ]

                # -------------------------------------------------
                # Extract columns explicitly mentioned by user
                # -------------------------------------------------

                extracted_columns = (
                    self.extract_visualization_columns(
                        context.message,
                        df.columns,
                    )
                )

                # =================================================
                # IDENTIFIER COLUMNS
                # =================================================

                identifier_columns = {
                    "id",
                    "passengerid",
                    "customerid",
                    "userid",
                    "user_id",
                    "index",
                    "rowid",
                    "recordid",
                    "record_id",
                }

                def is_identifier(column):

                    normalized = (
                        str(column)
                        .lower()
                        .replace(" ", "")
                        .replace("_", "")
                        .replace("-", "")
                    )

                    normalized_identifiers = {
                        str(name)
                        .lower()
                        .replace(" ", "")
                        .replace("_", "")
                        .replace("-", "")
                        for name in identifier_columns
                    }

                    return normalized in normalized_identifiers

                # =================================================
                # MEANINGFUL NUMERIC COLUMNS
                # =================================================

                numeric_columns = [
                    str(column)
                    for column in
                    df.select_dtypes(
                        include="number"
                    ).columns
                    if not is_identifier(column)
                ]

                # =================================================
                # CATEGORICAL COLUMNS
                # =================================================

                categorical_columns = [
                    str(column)
                    for column in
                    df.select_dtypes(
                        include=[
                            "object",
                            "category",
                            "bool",
                        ]
                    ).columns
                ]

                suitable_categorical_columns = []

                for column in categorical_columns:

                    unique_count = (
                        df[column]
                        .nunique(dropna=True)
                    )

                    if 2 <= unique_count <= 20:

                        suitable_categorical_columns.append(
                            column
                        )

                # =================================================
                # HISTOGRAM / BOXPLOT
                # =================================================

                if capability in {
                    "histogram",
                    "boxplot",
                }:

                    # Explicit user column wins
                    if extracted_columns.get("column"):

                        input_data["column"] = (
                            extracted_columns["column"]
                        )

                    else:

                        current_column = (
                            input_data.get("column")
                        )

                        # Never automatically use ID columns
                        if (
                            current_column
                            and is_identifier(
                                current_column
                            )
                        ):

                            input_data.pop(
                                "column",
                                None
                            )

                        # Automatic selection
                        if not input_data.get("column"):

                            preferred_columns = [
                                "Age",
                                "Fare",
                                "Pclass",
                                "SibSp",
                                "Parch",
                                "Survived",
                            ]

                            for column in preferred_columns:

                                if column in numeric_columns:

                                    input_data["column"] = (
                                        column
                                    )

                                    break

                            if not input_data.get(
                                "column"
                            ) and numeric_columns:

                                input_data["column"] = (
                                    numeric_columns[0]
                                )

                # =================================================
                # PIE CHART
                # =================================================

                elif capability == "pie_chart":

                    # Explicit user column
                    if extracted_columns.get("column"):

                        input_data["column"] = (
                            extracted_columns["column"]
                        )

                    # Validate LLM-selected column
                    elif (
                        input_data.get("column")
                        not in suitable_categorical_columns
                    ):

                        input_data.pop(
                            "column",
                            None
                        )

                    # Automatic categorical selection
                    if not input_data.get("column"):

                        preferred_categorical = [
                            "Sex",
                            "Embarked",
                            "Pclass",
                        ]

                        for column in preferred_categorical:

                            if (
                                column
                                in suitable_categorical_columns
                            ):

                                input_data["column"] = (
                                    column
                                )

                                break

                        if not input_data.get(
                            "column"
                        ) and suitable_categorical_columns:

                            input_data["column"] = (
                                suitable_categorical_columns[0]
                            )

                # =================================================
                # BAR CHART
                # =================================================

                elif capability == "bar_chart":

                    # Explicit X + Y
                    if (
                        extracted_columns.get("x")
                        and extracted_columns.get("y")
                    ):

                        input_data["x"] = (
                            extracted_columns["x"]
                        )

                        input_data["y"] = (
                            extracted_columns["y"]
                        )

                    # Explicit single column
                    elif extracted_columns.get("column"):

                        input_data["column"] = (
                            extracted_columns["column"]
                        )

                    else:

                        current_column = (
                            input_data.get("column")
                        )

                        # Reject unsuitable automatic column

                # =================================================

                    # Reject identifier columns
                    # -------------------------------------------------

                        input_data.get("x")
                        and is_identifier(
                            input_data["x"]
                        )
                    ):

                        input_data.pop(
                            "x",
                            None
                        )

                    if (
                        input_data.get("y")
                        and is_identifier(
                            input_data["y"]
                        )
                    ):

                        input_data.pop(
                            "y",
                            None
                        )

                    # -------------------------------------------------
                    # Automatic meaningful pair
                    # -------------------------------------------------

                    if (
                        not input_data.get("x")
                        or not input_data.get("y")
                    ):

                            ("Age", "Fare"),
                            ("Pclass", "Fare"),
                            ("Age", "SibSp"),
                            ("Age", "Parch"),
                        ]

                        for x_column, y_column in (
                            preferred_pairs
                        ):

                            if (
                                x_column in numeric_columns
                                and
                                y_column in numeric_columns
                            ):

                                selected_pair = (
                                    x_column,
                                    y_column,
                                )

                                break

                        if selected_pair:

                            if not input_data.get("x"):

                                input_data["x"] = (
                                    selected_pair[0]
                                )

                            if not input_data.get("y"):

                                input_data["y"] = (
                                    selected_pair[1]
                                )

                        elif len(numeric_columns) >= 2:

                            if not input_data.get("x"):

                                input_data["x"] = (
                                    numeric_columns[0]
                                )

                            if not input_data.get("y"):

                                input_data["y"] = (
                                    numeric_columns[1]
                                )

                # =================================================
                # LINE CHART
                # =================================================

                elif capability == "line_chart":

                    if extracted_columns.get("x"):

                        input_data["x"] = (

                    if extracted_columns.get("y"):

                        input_data.get("y")

                        )

                    # Automatic meaningful pair
                    if (
                        not input_data.get("x")
                        or not input_data.get("y")
                    ):

                        preferred_pairs = [
                            ("Age", "Fare"),
                            ("Pclass", "Fare"),
                            ("Age", "SibSp"),
                            ("Age", "Parch"),
                        ]

                        selected_pair = None

                        for x_column, y_column in (

                        ):

                                x_column in numeric_columns
                                and
                                y_column in numeric_columns
                            ):

                                selected_pair = (
                                    x_column,
                                    y_column,
                                )

                                break

                        if selected_pair:

                            if not input_data.get("x"):

                                )

                            if not input_data.get("x"):

                                input_data["x"] = (

                                )

                                    numeric_columns[1]
                                )

                # =================================================
                # HEATMAP
                # =================================================

                elif capability == "heatmap":

                    # Only meaningful numeric columns.
                    # Identifier columns such as PassengerId
                    # are deliberately excluded.

                    heatmap_columns = [
                        str(column)
                        for column in
                        df.select_dtypes(

        report_tasks = [
            task
            for task in plan.tasks
            if task.capability == "generate_report"
        ]

        if report_tasks:

            plan.tasks = [
                task
                for task in plan.tasks
                if task.capability != "generate_report"
            ]

            plan.tasks.extend(
                report_tasks
            )

        # =====================================================
        # SAVE PLAN
        # =====================================================

        context.execution_plan = plan

        return plan

        # =========================================================
    # EXTRACT VISUALIZATION COLUMNS
    # =========================================================

    @staticmethod
    def extract_visualization_columns(
        message: str,
        columns,
    ) -> dict:

        """
        Extract requested visualization columns directly
        from the user's message.

        Example:

        "Create a scatter plot using Age and Fare"

        returns:

        {
            "x": "Age",
            "y": "Fare"
        }
        """

        if not message:
            return {}

        if columns is None:
            return {}

        message_lower = message.lower()

        # -----------------------------------------------------
        # Create case-insensitive column lookup
        # -----------------------------------------------------

        column_lookup = {
            str(column).lower(): str(column)
            for column in columns
        }

        # -----------------------------------------------------
        # Find columns mentioned in the user request
        # -----------------------------------------------------

        mentioned_columns = []

        for column in columns:

            column_name = str(column)

            if column_name.lower() in message_lower:

                mentioned_columns.append(
                    column_name
                )

        # -----------------------------------------------------
        # Scatter / Bar / Line
        #
        # First mentioned column = X
        # Second mentioned column = Y
        # -----------------------------------------------------

        if len(mentioned_columns) >= 2:

            return {
                "x": mentioned_columns[0],
                "y": mentioned_columns[1],
            }

        # -----------------------------------------------------
        # One requested column
        # -----------------------------------------------------

        if len(mentioned_columns) == 1:

            return {
                "column": mentioned_columns[0]
            }

        return {}

    # PRIORITY
    # =========================================================

    @staticmethod
    def get_priority(
        priority: str
    ) -> TaskPriority:

        if not priority:
            return TaskPriority.MEDIUM

        priority = (
            priority
            .upper()
            .strip()
        )

        if priority == "HIGH":
            return TaskPriority.HIGH

        if priority == "LOW":
            return TaskPriority.LOW

        return TaskPriority.MEDIUM
